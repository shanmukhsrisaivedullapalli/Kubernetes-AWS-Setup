#!/bin/bash
set -euxo pipefail

# 1. Disable Swap & SELinux
swapoff -a
sed -i '/ swap / s/^\(.*\)$/#\1/g' /etc/fstab
setenforce 0 || true
sed -i 's/^SELINUX=enforcing$/SELINUX=permissive/' /etc/selinux/config

# 2. Kernel Modules & Sysctl Settings for Kubernetes Networking
cat <<EOF | tee /etc/modules-load.d/k8s.conf
overlay
br_netfilter
EOF

modprobe overlay
modprobe br_netfilter

cat <<EOF | tee /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-iptables  = 1
net.bridge.bridge-nf-call-ip6tables = 1
net.ipv4.ip_forward                 = 1
EOF

sysctl --system

# 3. Install and Configure containerd
yum install -y yum-utils
yum install -y containerd

mkdir -p /etc/containerd
containerd config default | tee /etc/containerd/config.toml
# Enable SystemdCgroup in containerd
sed -i 's/SystemdCgroup = false/SystemdCgroup = true/' /etc/containerd/config.toml

systemctl daemon-reload
systemctl enable --now containerd

# 4. Configure Kubernetes Package Repository (v1.31 example)
K8S_VERSION="v1.31"

cat <<EOF | tee /etc/yum.repos.d/kubernetes.repo
[kubernetes]
name=Kubernetes
baseurl=https://pkgs.k8s.io/core:/stable:/${K8S_VERSION}/rpm/
enabled=1
gpgcheck=1
gpgkey=https://pkgs.k8s.io/core:/stable:/${K8S_VERSION}/rpm/repodata/repomd.xml.key
exclude=kubelet kubeadm kubectl cri-tools kubernetes-cni
EOF

# Installing the prerequisites
dnf install -y conntrack iproute-tc iptables

# local hostname resolution
hostnamectl hostname K8s-Control-Plane
exec bash
echo "$(hostname -I | awk '{print $1}') $(hostname)" >> /etc/hosts

# 5. Install Kubernetes Tools
yum install -y kubelet kubeadm kubectl --disableexcludes=kubernetes
systemctl enable --now kubelet

# 6. Initialize Control Plane
# --pod-network-cidr is mandatory for standard CNIs like Calico (192.168.0.0/16) or Flannel (10.244.0.0/16)
kubeadm init --pod-network-cidr=192.168.0.0/16 --ignore-preflight-errors=NumCPU

# 7. Configure kubeconfig for root and regular users
mkdir -p /root/.kube
cp -i /etc/kubernetes/admin.conf /root/.kube/config
chown $(id -u):$(id -g) /root/.kube/config

# Optional: configure for ec2-user / cloud-user
if id "ec2-user" &>/dev/null; then
  mkdir -p /home/ec2-user/.kube
  cp -i /etc/kubernetes/admin.conf /home/ec2-user/.kube/config
  chown -R ec2-user:ec2-user /home/ec2-user/.kube
fi

# 8. Install CNI Plugin (Calico)
KUBECONFIG=/etc/kubernetes/admin.conf kubectl apply -f https://raw.githubusercontent.com/projectcalico/calico/v3.28.0/manifests/calico.yaml
